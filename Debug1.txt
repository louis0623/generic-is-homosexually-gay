local function makeHeaderDraggable(headerFrame, mainFrame)
    local dragging = false
    local dragInput, dragStart, startPos

    headerFrame.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            dragging = true
            dragStart = input.Position
            startPos = mainFrame.Position

            input.Changed:Connect(function()
                if input.UserInputState == Enum.UserInputState.End then
                    dragging = false
                end
            end)
        end
    end)

    headerFrame.InputChanged:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch then
            dragInput = input
        end
    end)

    UserInputService.InputChanged:Connect(function(input)
        if input == dragInput and dragging then
            local delta = input.Position - dragStart
            mainFrame.Position = UDim2.new(
                startPos.X.Scale,
                startPos.X.Offset + delta.X,
                startPos.Y.Scale,
                startPos.Y.Offset + delta.Y
            )
        end
    end)
end

local function playOpenAnim(frame, targetSize, targetPos)
    frame.Size = UDim2.new(0, 0, 0, 0)
    frame.Position = UDim2.new(targetPos.X.Scale, targetPos.X.Offset + (targetSize.X.Offset / 2), targetPos.Y.Scale, targetPos.Y.Offset + (targetSize.Y.Offset / 2))
    frame.Visible = true
    
    TweenService:Create(frame, TweenInfo.new(0.35, Enum.EasingStyle.Back, Enum.EasingDirection.Out), {
        Size = targetSize,
        Position = targetPos
    }):Play()
end

local function playCloseAnim(frame, targetSize, targetPos, onComplete)
    local centerPos = UDim2.new(targetPos.X.Scale, targetPos.X.Offset + (targetSize.X.Offset / 2), targetPos.Y.Scale, targetPos.Y.Offset + (targetSize.Y.Offset / 2))
    local tween = TweenService:Create(frame, TweenInfo.new(0.25, Enum.EasingStyle.Quad, Enum.EasingDirection.In), {
        Size = UDim2.new(0, 0, 0, 0),
        Position = centerPos
    })
    
    tween.Completed:Connect(function()
        frame.Visible = false
        frame.Size = targetSize
        frame.Position = targetPos
        if onComplete then onComplete() end
    end)
    tween:Play()
end

local function deactivateLine()
    if lineConnection then
        lineConnection:Disconnect()
        lineConnection = nil
    end
    if linePart then
        linePart:Destroy()
        linePart = nil
    end
    if currentESPButton then
        currentESPButton.Text = "Line Off"
        currentESPButton.BackgroundColor3 = Color3.fromRGB(70, 70, 80)
    end
    currentTarget = nil
    currentESPButton = nil
end

local function findMatches(query, catalogFilter, ignoreList)
    query = query:gsub("^%s+", ""):gsub("%s+$", "")
    if query == "" then return {} end
    
    local rootSearchScope = workspace
    if catalogFilter and catalogFilter ~= "" then
        for _, desc in workspace:GetDescendants() do
            if desc.Name:lower() == catalogFilter:lower() then
                rootSearchScope = desc
                break
            end
        end
    end

    local matches = {}
    for _, v in rootSearchScope:GetDescendants() do
        if (v:IsA("BasePart") or v:IsA("Model")) and v.Name:lower() == query:lower() then
            local isIgnored = false
            if ignoreList then
                for _, ignoredName in ipairs(ignoreList) do
                    if ignoredName ~= "" and v.Name:lower() == ignoredName:lower() then
                        isIgnored = true
                        break
                    end
                end
            end
            
            if not isIgnored then
                table.insert(matches, v)
            end
        end
    end
    return matches
end

local function getPosition(obj)
    if obj:IsA("BasePart") then return obj.Position end
    if obj:IsA("Model") then
        local part = obj.PrimaryPart or obj:FindFirstChildWhichIsA("BasePart")
        if part then return part.Position end
        return obj:GetBoundingBox().Position
    end
    return nil
end

local function getPath(obj)
    local names = {obj.Name}
    local parent = obj.Parent
    while parent and parent ~= workspace do
        table.insert(names, 1, parent.Name)
        parent = parent.Parent
    end
    return table.concat(names, " / ")
end

local function toggleLine(obj, button)
    if currentTarget == obj then
        deactivateLine()
        return
    end

    deactivateLine()
    currentTarget = obj
    currentESPButton = button

    linePart = Instance.new("Part")
    linePart.Anchored = true
    linePart.CanCollide = false
    linePart.CanQuery = false
    linePart.CastShadow = false
    linePart.Transparency = 0.15
    linePart.Color = Color3.fromRGB(255, 40, 40)
    linePart.Material = Enum.Material.Neon
    linePart.Size = Vector3.new(0.15, 0.15, 1)
    linePart.Parent = workspace

    button.Text = "Line On"
    button.BackgroundColor3 = Color3.fromRGB(0, 170, 90)

    lineConnection = RunService.RenderStepped:Connect(function()
        local char = player.Character
        local root = char and char:FindFirstChild("HumanoidRootPart")
        if not (root and currentTarget) then return end

        local bPos = getPosition(currentTarget)
        if not bPos then return end

        local aPos = root.Position
        local dist = (aPos - bPos).Magnitude
        local mid = (aPos + bPos) / 2

        linePart.CFrame = CFrame.lookAt(mid, bPos)
        linePart.Size = Vector3.new(0.15, 0.15, dist)
    end)
end

local function doSearch()
    deactivateLine()
    clearResults()
    status.Text = ""
    local matches = findMatches(inputBox.Text)

    if #matches == 0 then
        status.Text = "Nothing found"
        return
    end

    for _, obj in matches do
        addResult(obj)
    end
    status.Text = #matches .. " result(s)"
end

            attrLbl.TextXAlignment = Enum.TextXAlignment.Left
            attrLbl.TextTruncate = Enum.TextTruncate.AtEnd

    end

end

local function stopMonitoring()
    if activeNotifierThread then
        task.cancel(activeNotifierThread)
        activeNotifierThread = nil
    end
    knownParts = {}
    startNotifierBtn.Text = "Start Monitoring (5s Loop)"
    startNotifierBtn.BackgroundColor3 = Color3.fromRGB(0, 170, 90)
    notifierStatusLabel.Text = "Status: Idle"
end

startNotifierBtn.MouseButton1Click:Connect(function()
    if activeNotifierThread then
        stopMonitoring()
        return
    end

    local targetName = notifierInput.Text:gsub("^%s+", ""):gsub("%s+$", "")
    local catalogFilter = catalogInput.Text:gsub("^%s+", ""):gsub("%s+$", "")
    local rawIgnore = ignoreInput.Text
    
    local ignoreList = {}
    for partName in rawIgnore:gmatch("[^,]+") do
        local cleaned = partName:gsub("^%s+", ""):gsub("%s+$", "")
        if cleaned ~= "" then
            table.insert(ignoreList, cleaned)
        end
    end

    if targetName == "" then
        notifierStatusLabel.Text = "Status: Please enter a target name!"
        return
    end

    local initialMatches = findMatches(targetName, catalogFilter, ignoreList)
    if #initialMatches > 10 then
        notifierStatusLabel.Text = "warning! more than 10 items parts has detected,please use part searcher instead"
        return
    end

    startNotifierBtn.Text = "Stop Monitoring"
    startNotifierBtn.BackgroundColor3 = Color3.fromRGB(200, 60, 60)
    
    local filterMsg = catalogFilter ~= "" and ("\nCatalog: [" .. catalogFilter .. "]") or "\nCatalog: [Global Workspace]"
    local ignoreMsg = #ignoreList > 0 and ("\nIgnored: " .. table.concat(ignoreList, ", ")) or ""
    notifierStatusLabel.Text = "Monitoring: " .. targetName .. filterMsg .. ignoreMsg

    activeNotifierThread = task.spawn(function()
        local isFirstCheck = true
        while true do
            local currentMatches = findMatches(targetName, catalogFilter, ignoreList)
            
            if isFirstCheck then
                isFirstCheck = false
                if #currentMatches > 0 then
                    for _, item in ipairs(currentMatches) do
                        knownParts[item] = true
                    end
                    createNotification(#currentMatches .. " Item(s) already out there", currentMatches)
                end
            else
                local newlySpawned = {}
                for _, item in ipairs(currentMatches) do
                    if not knownParts[item] then
                        knownParts[item] = true
                        table.insert(newlySpawned, item)
                    end
                end

                if #newlySpawned > 0 then
                    createNotification("Item Spawned! (" .. #newlySpawned .. " new)", newlySpawned)
                end
            end

            task.wait(5)
        end
    end)
end)
