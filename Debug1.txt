-- ESP State
local lineConnection
local linePart
local currentTarget
local currentESPButton

local function deactivateLine()
    if lineConnection then
        lineConnection:Disconnect()
        lineConnection = nil
    end
    if linePart then
        linePart:Destroy()
        linePart = nil
    end
    if currentESPButton then
        currentESPButton.Text = "Line Off"
        currentESPButton.BackgroundColor3 = Color3.fromRGB(70, 70, 80)
    end
    currentTarget = nil
    currentESPButton = nil
end

local function findMatches(query)
    query = query:gsub("^%s+", ""):gsub("%s+$", "")
    if query == "" then return {} end
    local matches = {}
    for _, v in workspace:GetDescendants() do
        if (v:IsA("BasePart") or v:IsA("Model")) and v.Name:lower() == query:lower() then
            table.insert(matches, v)
        end
    end
    return matches
end

local function getPosition(obj)
    if obj:IsA("BasePart") then return obj.Position end
    if obj:IsA("Model") then
        local part = obj.PrimaryPart or obj:FindFirstChildWhichIsA("BasePart")
        if part then return part.Position end
        return obj:GetBoundingBox().Position
    end
    return nil
end

local function getPath(obj)
    local names = {obj.Name}
    local parent = obj.Parent
    while parent and parent ~= workspace do
        table.insert(names, 1, parent.Name)
        parent = parent.Parent
    end
    return table.concat(names, " / ")
end

local function toggleLine(obj, button)
    if currentTarget == obj then
        deactivateLine()
        return
    end

    deactivateLine()
    currentTarget = obj
    currentESPButton = button

    linePart = Instance.new("Part")
    linePart.Anchored = true
    linePart.CanCollide = false
    linePart.CanQuery = false
    linePart.CastShadow = false
    linePart.Transparency = 0.15
    linePart.Color = Color3.fromRGB(255, 40, 40)
    linePart.Material = Enum.Material.Neon
    linePart.Size = Vector3.new(0.15, 0.15, 1)
    linePart.Parent = workspace

    button.Text = "Line On"
    button.BackgroundColor3 = Color3.fromRGB(0, 170, 90)

    lineConnection = RunService.RenderStepped:Connect(function()
        local char = player.Character
        local root = char and char:FindFirstChild("HumanoidRootPart")
        if not (root and currentTarget) then return end

        local bPos = getPosition(currentTarget)
        if not bPos then return end

        local aPos = root.Position
        local dist = (aPos - bPos).Magnitude
        local mid = (aPos + bPos) / 2

        linePart.CFrame = CFrame.lookAt(mid, bPos)
        linePart.Size = Vector3.new(0.15, 0.15, dist)
    end)
end

local function clearResults()
    for _, child in listFrame:GetChildren() do
        if child:IsA("Frame") then child:Destroy() end
    end
end

local function doSearch()
    deactivateLine()
    clearResults()
    status.Text = ""
    local matches = findMatches(inputBox.Text)

    if #matches == 0 then
        status.Text = "Nothing found"
        return
    end

    for _, obj in matches do
        addResult(obj)
    end
    status.Text = #matches .. " result(s)"
end

searchBtn.MouseButton1Click:Connect(doSearch)
inputBox.FocusLost:Connect(function(enter)
    if enter then doSearch() end
end)

local function stopMonitoring()
    if activeNotifierThread then
        task.cancel(activeNotifierThread)
        activeNotifierThread = nil
    end
    knownParts = {}
    startNotifierBtn.Text = "Start Monitoring (5s Loop)"
    startNotifierBtn.BackgroundColor3 = Color3.fromRGB(0, 170, 90)
    notifierStatusLabel.Text = "Status: Idle"
end

startNotifierBtn.MouseButton1Click:Connect(function()
    if activeNotifierThread then
        stopMonitoring()
        return
    end

    local targetName = notifierInput.Text:gsub("^%s+", ""):gsub("%s+$", "")
    if targetName == "" then
        notifierStatusLabel.Text = "Status: Please enter a target name!"
        return
    end

    startNotifierBtn.Text = "Stop Monitoring"
    startNotifierBtn.BackgroundColor3 = Color3.fromRGB(200, 60, 60)
    notifierStatusLabel.Text = "Monitoring: " .. targetName

    activeNotifierThread = task.spawn(function()
        local isFirstCheck = true
        while true do
            local currentMatches = findMatches(targetName)
            
            if isFirstCheck then
                isFirstCheck = false
                if #currentMatches > 0 then
                    for _, item in ipairs(currentMatches) do
                        knownParts[item] = true
                    end
                    createNotification(#currentMatches .. " Item(s) already out there", currentMatches)
                end
            else
                local newlySpawned = {}
                for _, item in ipairs(currentMatches) do
                    if not knownParts[item] then
                        knownParts[item] = true
                        table.insert(newlySpawned, item)
                    end
                end

                if #newlySpawned > 0 then
                    createNotification("Item Spawned! (" .. #newlySpawned .. " new)", newlySpawned)
                end
            end

            task.wait(5)
        end
    end)
end)
